# HTML files CSS update script# 全HTMLファイルのCSS参照を新しい構造に更新$htmlFiles = Get-ChildItem -Path "C:\code" -Filter "*.html"foreach ($file in $htmlFiles) {    $content = Get-Content $file.FullName -Raw -Encoding UTF8    # 既存のstyle.css参照を新しいCSS構造に置換    $content = $content -replace '<link rel="stylesheet" href="style\.css"[^>]*>', @'<link rel="stylesheet" href="css/variables.css" />  <link rel="stylesheet" href="css/base.css" />  <link rel="stylesheet" href="css/layout.css" />  <link rel="stylesheet" href="css/components.css" />  <link rel="stylesheet" href="css/utilities.css" />'@    # スキップリンクを<body>直後に追加（まだない場合）    if ($content -notmatch 'class="skip-link"') {        $content = $content -replace '(<body[^>]*>)', '$1`n  <a href="#main" class="skip-link">メインコンテンツへスキップ</a>'    }    # headerにrole="banner"を追加    $content = $content -replace '<header>', '<header role="banner">'    # navにrole="navigation"とaria-labelを追加    $content = $content -replace '<nav>', '<nav role="navigation" aria-label="主要ナビゲーション">'    # footerにrole="contentinfo"を追加    $content = $content -replace '<footer>', '<footer role="contentinfo">'    # mainタグがある場合、role="main"を追加    $content = $content -replace '<main([^>]*)>', '<main$1 role="main">'    # mainタグがない場合、main-contentの親divをmainタグに変換    if ($content -notmatch '<main') {        $content = $content -replace '(<div class="container">[\r\n\s]*<div class="main-content">)', '<main id="main" role="main">`n    <div class="container">`n        <div class="main-content">'        $content = $content -replace '(</div>[\r\n\s]*</div>[\r\n\s]*<footer)', '</div>`n    </div>`n    </main>`n`n    <footer'    }    # モーダルにARIA属性を追加    if ($content -match '<div id="imageModal"') {        $content = $content -replace '<div id="imageModal" class="modal">', '<div id="imageModal" class="modal" role="dialog" aria-modal="true" aria-hidden="true" aria-labelledby="modal-title">'        # sr-onlyクラスのタイトルを追加        if ($content -notmatch 'id="modal-title"') {            $content = $content -replace '(<div id="imageModal"[^>]*>)', '$1`n        <h2 id="modal-title" class="sr-only">画像拡大表示</h2>'        }        # closeボタンをbuttonタグに変更してaria-labelを追加        $content = $content -replace '<span class="close">&times;</span>', '<button class="close" aria-label="モーダルを閉じる">&times;</button>'        # modal-contentのimgにaltを追加        $content = $content -replace '<img class="modal-content" id="img01">', '<img class="modal-content" id="img01" alt="拡大画像">'    }    # 更新されたコンテンツをファイルに書き戻す    Set-Content -Path $file.FullName -Value $content -Encoding UTF8 -NoNewline    Write-Host "Updated: $($file.Name)"}Write-Host "`nAll HTML files have been updated!"
